import { NextResponse, type NextRequest } from 'next/server';
import { z } from 'zod';
import { createRemoteJWKSet, jwtVerify, type JWTPayload } from 'jose';
import { createSessionForEmail } from '@/server/auth/otp';
import { errorResponse, HttpError, UnauthorizedError } from '@/server/auth/guards';

// jose needs Node crypto; keep this off the edge runtime.
export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

const BodySchema = z.object({
  provider: z.enum(['google', 'apple']),
  idToken: z.string().min(1),
  /** Apple only, and only on the user's first-ever authorization. */
  fullName: z.string().nullish(),
});

/**
 * The mobile client is not trusted. Anyone can POST this route with a
 * hand-written JWT claiming to be any email on earth, so the token's signature
 * is checked against the provider's published keys before a single claim is
 * read. `createRemoteJWKSet` lives at module scope on purpose: it caches and
 * rotates the key set, so this is one fetch per cold start rather than one per
 * sign-in (and it survives Google/Apple's periodic key rollover).
 */
const GOOGLE_JWKS = createRemoteJWKSet(
  new URL('https://www.googleapis.com/oauth2/v3/certs')
);
const APPLE_JWKS = createRemoteJWKSet(new URL('https://appleid.apple.com/auth/keys'));

/** Google issues `iss` in both forms depending on the client. Both are valid. */
const GOOGLE_ISSUERS = ['https://accounts.google.com', 'accounts.google.com'];
const APPLE_ISSUER = 'https://appleid.apple.com';

/**
 * Apple's audience is the app's bundle id, not an OAuth client id — the native
 * "Sign in with Apple" flow has no separate client registration.
 */
const APPLE_AUDIENCE = 'com.feraltravels.app';

interface VerifiedIdentity {
  email: string;
  name?: string;
}

function claimString(payload: JWTPayload, key: string): string | null {
  const value = payload[key];
  return typeof value === 'string' && value.length > 0 ? value : null;
}

/** Providers are inconsistent: `email_verified` arrives as boolean or "true". */
function claimIsTrue(payload: JWTPayload, key: string): boolean {
  const value = payload[key];
  return value === true || value === 'true';
}

async function verifyGoogle(idToken: string): Promise<VerifiedIdentity> {
  // The iOS client id — deliberately NOT the web AUTH_GOOGLE_ID. Accepting a
  // token minted for a different client would let any app holding a Google
  // token sign in as that user here (the "confused deputy" audience bug).
  const audience = process.env.GOOGLE_IOS_CLIENT_ID;
  if (!audience) {
    throw new HttpError(500, 'Configuration');
  }

  let payload: JWTPayload;
  try {
    // jwtVerify enforces the signature plus `exp` / `nbf`; issuer and audience
    // are checked here rather than after the fact so a bad token never reaches
    // the claim-reading code below.
    ({ payload } = await jwtVerify(idToken, GOOGLE_JWKS, {
      issuer: GOOGLE_ISSUERS,
      audience,
      clockTolerance: 5,
    }));
  } catch {
    throw new UnauthorizedError('InvalidToken');
  }

  const email = claimString(payload, 'email');
  if (!email) throw new UnauthorizedError('InvalidToken');

  // An unverified Google address is an address the holder may not own, and
  // this route's whole contract is "verified email == account identity".
  if (!claimIsTrue(payload, 'email_verified')) {
    throw new UnauthorizedError('EmailNotVerified');
  }

  return { email: email.toLowerCase(), name: claimString(payload, 'name') ?? undefined };
}

async function verifyApple(
  idToken: string,
  fullName: string | null | undefined
): Promise<VerifiedIdentity> {
  let payload: JWTPayload;
  try {
    ({ payload } = await jwtVerify(idToken, APPLE_JWKS, {
      issuer: APPLE_ISSUER,
      audience: APPLE_AUDIENCE,
      clockTolerance: 5,
    }));
  } catch {
    throw new UnauthorizedError('InvalidToken');
  }

  const email = claimString(payload, 'email');
  if (!email) throw new UnauthorizedError('InvalidToken');

  // Apple's token carries no name claim ever, and the client can only send one
  // on the user's first authorization — so this is the single chance to record
  // it. It is display-only and never used for identity.
  const name = fullName?.trim() || undefined;

  return { email: email.toLowerCase(), name };
}

export async function POST(req: NextRequest) {
  try {
    const body = BodySchema.parse(await req.json());

    const identity =
      body.provider === 'google'
        ? await verifyGoogle(body.idToken)
        : await verifyApple(body.idToken, body.fullName);

    // Same helper the OTP route uses — identical session rows either way.
    const session = await createSessionForEmail(identity.email, identity.name);

    return NextResponse.json({
      token: session.token,
      expires: session.expires,
      user: { id: session.user.id, email: session.user.email },
    });
  } catch (err) {
    if (err instanceof z.ZodError) {
      return NextResponse.json({ error: 'InvalidRequest' }, { status: 400 });
    }
    return errorResponse(err);
  }
}
